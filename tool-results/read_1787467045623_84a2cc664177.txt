     1→/**
     2→ * SheetsShellCoordinator — owns the per-renderer workbook session lifecycle.
     3→ *
     4→ * RESOURCE OWNERSHIP (Increment 4D/4E/4F/4G):
     5→ *   OwnedResources is created BEFORE the first resource. The operation owns
     6→ *   every resource from creation until transfer() or release().
     7→ *   Conversion temp dirs are owned: cleaned up eagerly after snapshot creation,
     8→ *   but if cleanup fails, ownership is RETAINED and release() retries.
     9→ *
    10→ * SAVE COMMIT PROTOCOL (Increment 4F/4G):
    11→ *   Phase A — Prepare: temp target + snapshot + open + validate.
    12→ *   Phase B — Commit: transition to COMMITTING → write marker → rename →
    13→ *             install replacement → clear marker → transition to IDLE.
    14→ *             If rename fails, save fails explicitly — NO copyFile fallback.
    15→ *   Phase C — Cleanup: close old handle, remove old snapshot (best-effort).
    16→ *
    17→ * SESSION COMMIT LIFECYCLE (Increment 4G):
    18→ *   Each session has a commit state: IDLE | COMMITTING | TEARING_DOWN | CLOSED.
    19→ *   - Teardown checks: if session is COMMITTING, teardown waits for the
    20→ *     commit to complete (it cannot preempt an in-progress commit).
    21→ *   - Save checks: if session is TEARING_DOWN, save aborts before commit.
    22→ *   This gives deterministic semantics: teardown cannot corrupt a committed save.
    23→ *
    24→ * CRASH RECONCILIATION (Increment 4G):
    25→ *   Commit markers are stored in a DETERMINISTIC userData directory:
    26→ *     userData/sheets-save-commits/<sessionId>.json
    27→ *   Both save and reconcileSaveCommit() use the SAME location.
    28→ *   Markers are validated at read time (not unchecked casts).
    29→ */

    88→// ── OwnedResources ──
    89→
    90→class OwnedResources {
    91→  private snapshotPath: string | undefined
    92→  private engineHandle: EngineSessionHandle | undefined
    93→  private tempTargetPath: string | undefined
    94→  private conversionDir: string | undefined
    95→  private _transferred = false
    96→
    97→  setSnapshot(path: string): void { this.snapshotPath = path }
    98→  setEngineHandle(handle: EngineSessionHandle): void { this.engineHandle = handle }
    99→  setTempTarget(path: string): void { this.tempTargetPath = path }
   100→  setConversionDir(dir: string): void { this.conversionDir = dir }
   101→  clearConversionDir(): void { this.conversionDir = undefined }
   102→  clearTempTarget(): void { this.tempTargetPath = undefined }
   103→  get tempTarget(): string | undefined { return this.tempTargetPath }
   104→  get transferred(): boolean { return this._transferred }
   105→
   106→  async release(service: SpreadsheetService): Promise<void> {
   107→    if (this._transferred) return
   108→    if (this.engineHandle) { try { await service.close(this.engineHandle) } catch {} this.engineHandle = undefined }
   109→    if (this.snapshotPath) { try { await rm(this.snapshotPath, { force: true }) } catch {} this.snapshotPath = undefined }
   110→    if (this.tempTargetPath) { try { await rm(this.tempTargetPath, { force: true }) } catch {} this.tempTargetPath = undefined }
   111→    if (this.conversionDir) { try { await rm(this.conversionDir, { recursive: true, force: true }) } catch {} this.conversionDir = undefined }
   112→  }
   113→
   114→  transfer(): void {
   115→    this.snapshotPath = undefined; this.engineHandle = undefined
   116→    this.tempTargetPath = undefined; this.conversionDir = undefined
   117→    this._transferred = true
   118→  }
   119→}

   121→// ── Save commit marker ──
   122→
   123→interface SaveCommitMarker {
   124→  readonly version: 1
   125→  readonly finalTarget: string
   126→  readonly tempTarget: string
   127→  readonly sessionId: string
   128→}
   129→
   130→/** Type guard: is the value a non-null, non-array object (a record)? */
   131→function isRecord(value: unknown): value is Record<string, unknown> {
   132→  return typeof value === 'object' && value !== null && !Array.isArray(value)
   133→}
   134→
   135→/**
   136→ * Validate a parsed JSON value as a SaveCommitMarker.
   137→ * Uses only runtime type guards — no `as` type assertions.
   138→ * Returns null if the value is not a valid marker.
   139→ */
   140→function validateMarker(raw: unknown): SaveCommitMarker | null {
   141→  if (!isRecord(raw)) return null
   142→  if (raw.version !== 1) return null
   143→  if (typeof raw.finalTarget !== 'string' || raw.finalTarget.length === 0) return null
   144→  if (typeof raw.tempTarget !== 'string' || raw.tempTarget.length === 0) return null
   145→  if (typeof raw.sessionId !== 'string' || raw.sessionId.length === 0) return null
   146→  return { version: 1, finalTarget: raw.finalTarget, tempTarget: raw.tempTarget, sessionId: raw.sessionId }
   147→}

   149→// ── Coordinator ──

   151→export class SheetsShellCoordinator {
   152→  private readonly tabs = new Map<number, RendererState>()
   153→
   154→  constructor(private readonly deps: SheetsShellCoordinatorDeps) {}
   155→
   156→  registerRenderer(wcId: number, webContents: WebContents): void {
   157→    if (!this.tabs.has(wcId)) {
   158→      this.tabs.set(wcId, {
   159→        sessions: new Map(), epoch: 0, locks: new Map(), commitStates: new Map(),
   160→      })
   161→    }
   162→    webContents.once('destroyed', () => { void this.teardown(wcId) })
   163→  }
   164→
   165→  getSession(wcId: number, sessionId: string): ShellWorkbookSession {
   166→    const state = this.tabs.get(wcId)
   167→    if (!state) throw new InvalidSessionError(`Unknown renderer: ${wcId}`)
   168→    const session = state.sessions.get(sessionId)
   169→    if (!session) throw new InvalidSessionError(`Unknown workbook session: ${sessionId}`)
   170→    return session
   171→  }
   172→
   173→  wcIdFromEvent(event: IpcMainInvokeEvent): number { return event.sender.id }
   174→
   175→  private isAlive(wcId: number, startEpoch: number): boolean {
   176→    const state = this.tabs.get(wcId)
   177→    return !!state && state.epoch === startEpoch
   178→  }
   179→
   180→  private checkEpoch(wcId: number, startEpoch: number): void {
   181→    if (!this.isAlive(wcId, startEpoch)) throw new InvalidSessionError(`Renderer ${wcId} was torn down during operation`)
   182→  }
   183→
   184→  private withSessionLock<T>(wcId: number, sessionId: string, fn: () => Promise<T>): Promise<T> {
   185→    const state = this.tabs.get(wcId)
   186→    if (!state) throw new InvalidSessionError(`Unknown renderer: ${wcId}`)
   187→    const prev = state.locks.get(sessionId) ?? Promise.resolve()
   188→    const next = prev.then(() => fn(), () => fn())
   189→    state.locks.set(sessionId, next.catch(() => {}))
   190→    return next
   191→  }
   192→
   193→  private getCommitState(wcId: number, sessionId: string): SessionCommitState {
   194→    return this.tabs.get(wcId)?.commitStates.get(sessionId) ?? SessionCommitState.IDLE
   195→  }
   196→
   197→  private setCommitState(wcId: number, sessionId: string, state: SessionCommitState): void {
   198→    this.tabs.get(wcId)?.commitStates.set(sessionId, state)
   199→  }

   201→  // ── Open ──
   202→
   203→  async openWorkbook(
   204→    wcId: number, callerWindow: BrowserWindow | undefined,
   205→    options: { queuedPath?: string | undefined; locale: string },
   206→  ): Promise<{ sessionId: string; session: ShellWorkbookSession } | null> {
   207→    let state = this.tabs.get(wcId)
   208→    if (!state) {
   209→      state = { sessions: new Map(), epoch: 0, locks: new Map(), commitStates: new Map() }
   210→      this.tabs.set(wcId, state)
   211→    }
   212→    const startEpoch = state.epoch
   213→
   214→    let path = options.queuedPath
   215→    if (!path) {
   216→      const selection = callerWindow
   217→        ? await dialog.showOpenDialog(callerWindow, { properties: ['openFile'], filters: [{ name: 'Spreadsheets', extensions: ['xlsx', 'xls', 'csv'] }] })
   218→        : await dialog.showOpenDialog({ properties: ['openFile'], filters: [{ name: 'Spreadsheets', extensions: ['xlsx', 'xls', 'csv'] }] })
   219→      this.checkEpoch(wcId, startEpoch)
   220→      if (selection.canceled || !selection.filePaths[0]) return null
   221→      path = selection.filePaths[0]
   222→    }
   223→
   224→    const prepared = await this.prepareWorkbookForOpen(path, callerWindow)
   225→    this.checkEpoch(wcId, startEpoch)
   226→
   227→    const owned = new OwnedResources()
   228→    if (prepared.conversionDir) owned.setConversionDir(prepared.conversionDir)
   229→
   230→    try {
   231→      const snapshotPath = await this.snapshotWorkbook(prepared.openPath)
   232→      owned.setSnapshot(snapshotPath)
   233→      this.checkEpoch(wcId, startEpoch)
   234→
   235→      if (prepared.conversionDir) {
   236→        try {
   237→          await rm(prepared.conversionDir, { recursive: true, force: true })
   238→          owned.clearConversionDir()
   239→        } catch { /* retained for release() retry */ }
   240→      }
   241→
   242→      const bytes = await readFile(snapshotPath)
   243→      const fileName = prepared.openPath.split(/[\\/]/).pop() ?? 'workbook.xlsx'
   244→      const openResult = await this.deps.service.open(new Uint8Array(bytes), options.locale, fileName)
   245→      owned.setEngineHandle(openResult.engineHandle)
   246→      this.checkEpoch(wcId, startEpoch)
   247→
   248→      const diskFingerprint = await this.sha256File(snapshotPath)
   249→      const restoreTargetSha = prepared.restoreTarget
   250→        ? await this.sha256File(prepared.restoreTarget).catch(() => undefined) : undefined
   251→
   252→      this.checkEpoch(wcId, startEpoch)
   253→      const sessionId = randomUUID()
   254→      const shellSession: ShellWorkbookSession = {
   255→        sessionId, originalPath: path, snapshotPath, diskFingerprint,
   256→        suggestSaveAs: prepared.suggestSaveAs, csvImport: prepared.csvImport,
   257→        restoreTarget: prepared.restoreTarget, restoreTargetSha,
   258→        engineHandle: openResult.engineHandle, domainSession: openResult.session,
   259→        metadata: openResult.metadata, locale: options.locale, recoveryEpoch: 0,
   260→      }
   261→      state = this.tabs.get(wcId)!
   262→      state.sessions.set(sessionId, shellSession)
   263→      state.commitStates.set(sessionId, SessionCommitState.IDLE)
   264→      owned.transfer()
   265→
   266→      return { sessionId, session: shellSession }
   267→    } catch (error) {
   268→      await owned.release(this.deps.service)
   269→      throw error
   270→    }
   271→  }

   273→  // ── Legacy session adoption (Increment 5A) ──
   274→
   275→  /**
   276→   * Adopt a legacy-opened session into the coordinator's registry.
   277→   *
   278→   * This is the compatibility handoff between the legacy `workbook:select`
   279→   * open lifecycle (which uses `XlsxSidecarClient.open()` directly) and the
   280→   * migrated read/recalc/media/close path (which uses the coordinator-backed
   281→   * `SpreadsheetService`).
   282→   *
   283→   * The caller (sheets-runtime.ts) has ALREADY:
   284→   *   - Opened the workbook via the legacy `XlsxSidecarClient.open()`.
   285→   *   - Built a snapshot, computed its sha256, captured sheet names.
   286→   *   - Constructed a `ShellWorkbookSession` whose `engineHandle` was produced
   287→   *     by `ElectronXlsxSidecarEngine.adoptExternalSession()` — a pure
   288→   *     in-process handle wrap with NO wire call, NO file IO, NO spawn.
   289→   *
   290→   * The coordinator just registers the session under (wcId, sessionId) and
   291→   * initializes its commit state to IDLE. There is NO re-open, NO re-spawn,
   292→   * NO duplicate snapshot, NO recomputed fingerprint.
   293→   *
   294→   * OWNERSHIP: After adoption, the coordinator is the EXCLUSIVE owner of:
   295→   *   - The engine handle (and the underlying sidecar session it wraps).
   296→   *   - The snapshot path.
   297→   *   - The recovery resources.
   298→   * The legacy `SessionInfo` keeps a NON-OWNING reference (marked
   299→   * `adopted: true`) for the legacy `save`/`write-recovery` paths to keep
   300→   * working — but it MUST NOT independently `client.close(sessionId)` or
   301→   * `rm(snapshotPath)` once adopted.
   302→   *
   303→   * SESSION IDENTITY: The coordinator registers the session under the SAME
   304→   * `sessionId` the legacy open returned — preserving renderer continuity.
   305→   * The renderer's existing `sessionId` is valid for both legacy
   306→   * `save`/`write-recovery` AND migrated reads/closes without remapping.
   307→   *
   308→   * THREAD SAFETY: Acquires the per-session mutation lock. If a concurrent
   309→   * teardown already holds the session's lock (e.g., the renderer is being
   310→   * destroyed), adoption still completes — the session is registered but
   311→   * will be cleaned up by the teardown that owns the lock.
   312→   *
   313→   * @returns the registered session (same as the input).
   314→   */
   315→  async adoptLegacySession(
   316→    wcId: number,
   317→    session: ShellWorkbookSession,
   318→  ): Promise<ShellWorkbookSession> {
   319→    // Lazily register the renderer if this is the first call for this wcId.
   320→    // The legacy `workbook:select` path does not call registerRenderer()
   321→    // explicitly — adoption is the first coordinator contact for a tab.
   322→    let state = this.tabs.get(wcId)
   323→    if (!state) {
   324→      state = {
   325→        sessions: new Map(),
   326→        epoch: 0,
   327→        locks: new Map(),
   328→        commitStates: new Map(),
   329→      }
   330→      this.tabs.set(wcId, state)
   331→    }
   332→    return this.withSessionLock(wcId, session.sessionId, async () => {
   333→      // Re-fetch inside the lock — the renderer may have been torn down
   334→      // between the lazy register above and the lock acquisition.
   335→      const cur = this.tabs.get(wcId)
   336→      if (!cur) {
   337→        throw new InvalidSessionError(`Renderer ${wcId} torn down during adoption`)
   338→      }
   339→      // If a session already exists under this sessionId, it means adoption
   340→      // was called twice (a bug, or a save-reopen path that hasn't been
   341→      // migrated yet). The previous session's resources must be cleaned up
   342→      // by the caller BEFORE re-adoption — we throw to surface the bug.
   343→      if (cur.sessions.has(session.sessionId)) {
   344→        throw new InvalidSessionError(
   345→          `Session ${session.sessionId} already adopted — caller must close before re-adopting`,
   346→        )
   347→      }
   348→      cur.sessions.set(session.sessionId, session)
   349→      cur.commitStates.set(session.sessionId, SessionCommitState.IDLE)
   350→      return session
   351→    })
   352→  }

   373→  // ── Save (commit protocol with lifecycle state) ──
   374→
   375→  async saveWorkbook(
   376→    wcId: number, sessionId: string, request: SaveRequest, mode: 'save' | 'save-as',
   377→    callerWindow: BrowserWindow | undefined,
   378→  ): Promise<SaveResult & { canceled?: boolean }> {
   379→    return this.withSessionLock(wcId, sessionId, async () => {
   380→      const state = this.tabs.get(wcId)
   381→      if (!state) throw new InvalidSessionError(`Unknown renderer: ${wcId}`)
   382→      const startEpoch = state.epoch
   383→      const session = this.getSession(wcId, sessionId)
   384→
   385→      // Check commit state — if teardown is in progress, abort
   386→      if (this.getCommitState(wcId, sessionId) === SessionCommitState.TEARING_DOWN) {
   387→        throw new InvalidSessionError(`Session ${sessionId} is being torn down`)
   388→      }
   389→
   390→      // 1. Resolve target + ExternalChangeStatus
   391→      let targetPath: string
   392→      let externalChange: ExternalChangeStatus
   393→      if (mode === 'save-as' || session.suggestSaveAs !== undefined) {
   394→        const opts = { defaultPath: session.suggestSaveAs ?? session.restoreTarget ?? session.originalPath, filters: [{ name: 'Excel', extensions: ['xlsx'] }] }
   395→        const sel = callerWindow ? await dialog.showSaveDialog(callerWindow, opts) : await dialog.showSaveDialog(opts)
   396→        this.checkEpoch(wcId, startEpoch)
   397→        if (sel.canceled || !sel.filePath) return { ok: false, canceled: true }
   398→        targetPath = sel.filePath.endsWith('.xlsx') ? sel.filePath : `${sel.filePath}.xlsx`
   399→        externalChange = 'unchanged'
   400→      } else if (session.restoreTarget !== undefined) {
   401→        externalChange = await this.computeExternalChangeStatus(session.restoreTarget, session.restoreTargetSha ?? '')
   402→        this.checkEpoch(wcId, startEpoch)
   403→        targetPath = session.restoreTarget
   404→      } else {
   405→        externalChange = await this.computeExternalChangeStatus(session.originalPath, session.diskFingerprint)
   406→        this.checkEpoch(wcId, startEpoch)
   407→        targetPath = session.originalPath
   408→      }
   409→
   410→      // 2. service.save()
   411→      const result = await this.deps.service.save(session.domainSession, session.engineHandle, request, externalChange)
   412→      this.checkEpoch(wcId, startEpoch)
   413→      if (!result.ok || !result.data) return result
   414→
   415→      // ═══ Phase A: Prepare ═══
   416→      const owned = new OwnedResources()
   417→      let replacementSession: ShellWorkbookSession
   418→      let tempTargetPath: string
   419→      try {
   420→        tempTargetPath = join(dirname(targetPath), `.genoffice-save-${randomUUID()}.xlsx`)
   421→        await writeFile(tempTargetPath, result.data)
   422→        owned.setTempTarget(tempTargetPath)
   423→        this.checkEpoch(wcId, startEpoch)
   424→
   425→        const newSnapshotPath = await this.snapshotWorkbook(tempTargetPath)
   426→        owned.setSnapshot(newSnapshotPath)
   427→        this.checkEpoch(wcId, startEpoch)
   428→
   429→        const newBytes = await readFile(newSnapshotPath)
   430→        const fileName = targetPath.split(/[\\/]/).pop() ?? 'workbook.xlsx'
   431→        const newOpenResult = await this.deps.service.open(new Uint8Array(newBytes), session.locale, fileName)
   432→        owned.setEngineHandle(newOpenResult.engineHandle)
   433→        this.checkEpoch(wcId, startEpoch)
   434→
   435→        const newDiskFingerprint = await this.sha256File(newSnapshotPath)
   436→        this.checkEpoch(wcId, startEpoch)
   437→
   438→        replacementSession = {
   439→          sessionId, originalPath: targetPath, snapshotPath: newSnapshotPath,
   440→          diskFingerprint: newDiskFingerprint, engineHandle: newOpenResult.engineHandle,
   441→          domainSession: newOpenResult.session, metadata: newOpenResult.metadata,
   442→          locale: session.locale, recoveryEpoch: session.recoveryEpoch + 1,
   443→        }
   444→      } catch (error) {
   445→        await owned.release(this.deps.service)
   446→        throw error
   447→      }
   448→
   449→      // ═══ Phase B: Commit ═══
   450→      // Check epoch AND commit state before entering COMMITTING.
   451→      // If teardown has already started, abort — do NOT commit.
   452→      this.checkEpoch(wcId, startEpoch)
   453→      if (this.getCommitState(wcId, sessionId) === SessionCommitState.TEARING_DOWN) {
   454→        await owned.release(this.deps.service)
   455→        throw new InvalidSessionError(`Session ${sessionId} is being torn down`)
   456→      }
   457→
   458→      // Transition to COMMITTING — teardown that arrives after this
   459→      // MUST wait for the commit to complete (it cannot preempt).
   460→      this.setCommitState(wcId, sessionId, SessionCommitState.COMMITTING)
   461→
   462→      // Commit gate — injectable barrier for deterministic testing.
   463→      // In production, this is a no-op. In tests, it pauses the save
   464→      // between COMMITTING and the irreversible rename, allowing the
   465→      // test to verify the state and trigger teardown.
   466→      if (this.deps.onCommitGate) {
   467→        await this.deps.onCommitGate(sessionId)
   468→      }
   469→
   470→      try {
   471→        // Write commit marker to the DETERMINISTIC userData directory
   472→        // (same location reconcileSaveCommit scans)
   473→        const commitDir = this.commitMarkerDir()
   474→        await mkdir(commitDir, { recursive: true })
   475→        const markerPath = join(commitDir, `${sessionId}.json`)
   476→        const marker: SaveCommitMarker = { version: 1, finalTarget: targetPath, tempTarget: tempTargetPath, sessionId }
   477→        await writeFile(markerPath, JSON.stringify(marker))
   478→
   479→        // Marker-written hook — injectable barrier for deterministic
   480→        // testing of the save→reconcile crash path. Called AFTER the
   481→        // marker is on disk but BEFORE the rename. In production, undefined.
   482→        if (this.deps.onMarkerWritten) {
   483→          await this.deps.onMarkerWritten(markerPath, sessionId)
   484→        }
   485→
   486→        // Atomically promote temp → final via rename.
   487→        // NO copyFile fallback. If rename fails, save fails.
   488→        try {
   489→          await rename(tempTargetPath, targetPath)
   490→        } catch (renameError) {
   491→          try { await rm(markerPath, { force: true }) } catch {}
   492→          throw new EngineError(
   493→            `Save commit failed: cannot atomically rename temp to final target — ${renameError}`,
   494→            'INTERNAL_ERROR',
   495→          )
   496→        }
   497→        owned.clearTempTarget()
   498→
   499→        // Install replacement session
   500→        const currentState = this.tabs.get(wcId)
   501→        if (currentState) currentState.sessions.set(sessionId, replacementSession)
   502→        owned.transfer()
   503→
   504→        // Clear commit marker
   505→        try { await rm(markerPath, { force: true }) } catch {}
   506→
   507→        // Clear recovery copies
   508→        this.clearWorkbookRecovery(targetPath)
   509→        if (session.suggestSaveAs !== undefined) this.clearWorkbookRecovery(session.suggestSaveAs)
   510→        if (session.restoreTarget !== undefined) this.clearWorkbookRecovery(session.restoreTarget)
   511→      } catch (error) {
   512→        // Commit failure — release owned resources
   513→        await owned.release(this.deps.service)
   514→        this.setCommitState(wcId, sessionId, SessionCommitState.IDLE)
   515→        throw error
   516→      }
   517→
   518→      // ═══ Phase C: Old-resource cleanup (isolated, best-effort) ═══
   519→      this.setCommitState(wcId, sessionId, SessionCommitState.IDLE)
   520→      try { await this.deps.service.close(session.engineHandle) } catch {}
   521→      try { await rm(session.snapshotPath, { force: true }) } catch {}
   522→
   523→      return result
   524→    })
   525→  }

   527→  // ── Crash reconciliation ──
   528→
   529→  /**
   530→   * Reconcile leftover save-commit markers from a crash during save.
   531→   *
   532→   * Markers are stored in: userData/sheets-save-commits/<sessionId>.json
   533→   *
   534→   * For each marker:
   535→   *   - If tempTarget exists: delete it (rename never happened or crashed after)
   536→   *   - Delete the marker
   537→   *
   538→   * If final target already has new bytes (rename succeeded before crash),
   539→   * the old bytes are already gone — we can't undo the rename. But the
   540→   * temp file is cleaned up.
   541→   */
   542→  static async reconcileSaveCommit(userDataDir: string): Promise<void> {
   543→    const commitDir = join(userDataDir, 'sheets-save-commits')
   544→    let entries: string[]
   545→    try { entries = await readdir(commitDir) } catch { return }
   546→
   547→    for (const entry of entries) {
   548→      if (!entry.endsWith('.json')) continue
   549→      const markerPath = join(commitDir, entry)
   550→      try {
   551→        const markerText = await readFile(markerPath, 'utf8')
   552→        const raw = JSON.parse(markerText)
   553→        const marker = validateMarker(raw)
   554→        if (!marker) {
   555→          // Invalid marker — delete it
   556→          try { await rm(markerPath, { force: true }) } catch {}
   557→          continue
   558→        }
   559→        // Clean up temp target if it still exists
   560→        if (existsSync(marker.tempTarget)) {
   561→          try { await rm(marker.tempTarget, { force: true }) } catch {}
   562→        }
   563→        // Always clean up the marker
   564→        try { await rm(markerPath, { force: true }) } catch {}
   565→      } catch {
   566→        // Read/parse failed — delete the marker
   567→        try { await rm(markerPath, { force: true }) } catch {}
   568→      }
   569→    }
   570→  }

   627→  // ── Auto-rename (INCREMENT 12, hardened 15) ──
   628→
   629→  /**
   630→   * Rename the workbook file on disk.
   631→   *
   632→   * The coordinator owns:
   633→   *   - session lookup (wcId + sessionId → ShellWorkbookSession)
   634→   *   - rename validation (untitled path check, name sanitization, collision)
   635→   *   - filesystem rename (renameSync)
   636→   *   - session path update (ShellWorkbookSession.originalPath = newPath)
   637→   *   - renderer notification (workbook:renamed push to event.sender only)
   638→   *
   639→   * SESSION CONTINUITY:
   640→   *   - sessionId remains unchanged
   641→   *   - engineHandle remains unchanged
   642→   *   - snapshotPath remains unchanged (the snapshot is independent of the file path)
   643→   *   - originalPath becomes the new path
   644→   *   - diskFingerprint remains correct (the file content didn't change)
   645→   *
   646→   * PUSH EVENT:
   647→   *   workbook:renamed is sent ONLY to the initiating renderer (event.sender).
   648→   *   No broadcast. No getFocusedWindow.
   649→   *
   650→   * LEGACY COMPATIBILITY:
   651→   *   The legacy SessionInfo.path is updated via the shell's compatibility mirror
   652→   *   (sheetsTabs). The coordinator's ShellWorkbookSession.originalPath is the
   653→   *   AUTHORITATIVE owner after migration.
   654→   *
   655→   * @returns { renamed: boolean; name?: string }
   656→   */
   657→  async renameWorkbook(
   658→    wcId: number,
   659→    webContents: WebContents,
   660→    sessionId: string,
   661→    baseName: string,
   662→  ): Promise<{ renamed: boolean; name?: string }> {
   663→    return this.withSessionLock(wcId, sessionId, async () => {
   664→      // Re-fetch the session inside the lock — it may have been modified
   665→      // by a concurrent save or close.
   666→      let session: ShellWorkbookSession
   667→      try {
   668→        session = this.getSession(wcId, sessionId)
   669→      } catch {
   670→        return { renamed: false }
   671→      }
   672→
   673→      // Only rename untitled workbooks (matching legacy: untitledWorkbookPaths check)
   674→      if (!this.isUntitledPath(session.originalPath)) {
   675→        return { renamed: false }
   676→      }
   677→
   678→      // Sanitize the base name
   679→      const base = sanitizeAutoRenameBase(baseName)
   680→      if (!base) return { renamed: false }
   681→
   682→      // Compute target path (collision avoidance: name-2, name-3, ...)
   683→      const dir = dirname(session.originalPath)
   684→      let target = join(dir, `${base}.xlsx`)
   685→      for (let i = 2; existsSync(target) && i < 100; i++) {
   686→        target = join(dir, `${base}-${i}.xlsx`)
   687→      }
   688→      if (existsSync(target) || target === session.originalPath) {
   689→        return { renamed: false }
   690→      }
   691→
   692→      // Atomic rename (no copy fallback — matching legacy)
   693→      try {
   694→        renameSync(session.originalPath, target)
   695→      } catch (err) {
   696→        console.warn('[sheets] auto-rename failed:', err)
   697→        return { renamed: false }
   698→      }
   699→
   700→      // Update the coordinator's session (authoritative owner)
   701→      const state = this.tabs.get(wcId)
   702→      if (state) {
   703→        const updatedSession: ShellWorkbookSession = {
   704→          ...session,
   705→          originalPath: target,
   706→        }
   707→        state.sessions.set(sessionId, updatedSession)
   708→      }
   709→
   710→      // Remove from untitled set
   711→      this.untitledPaths.delete(session.originalPath)
   712→
   713→      // Push event to the initiating renderer only
   714→      const name = basename(target)
   715→      if (!webContents.isDestroyed()) {
   716→        webContents.send('workbook:renamed', name)
   717→      }
   718→
   719→      return { renamed: true, name }
   720→    })
   721→  }

   736→  // ── PDF export (INCREMENT 7 / ADR-006) ──
   737→
   738→  /**
   739→   * Export the workbook to PDF.
   740→   *
   741→   * Returns:
   742→   *   - { canceled: true } — user canceled the save dialog
   743→   *   - { canceled: false, path: string } — success, PDF written to path
   744→   *
   745→   * Errors (render failure, filesystem failure) are THROWN — matching the
   746→   * legacy exportPdf behavior.
   747→   *
   748→   * The coordinator owns callerWindow + save dialog + output authorization +
   749→   * writing the PDF bytes. The PDF renderer (SpreadsheetPdfRenderer) owns
   750→   * only the rendering context (hidden BrowserWindow + printToPDF + cleanup).
   751→   *
   752→   * SECURITY ORDERING (output authorization):
   753→   *   1. Show save dialog → user selects path (or cancels)
   754→   *   2. If canceled → return { canceled: true }
   755→   *   3. Render HTML → PDF bytes (via the renderer port)
   756→   *   4. If render fails → throw Error (no file written)
   757→   *   5. Write PDF bytes to the authorized path
   758→   *
   759→   * INCREMENT 7A: test-only output path override. When the environment
   760→   * variable `GENOFFICE_PDF_TEST_OUTPATH` is set, the save dialog is
   761→   * SKIPPED and the env var's value is used as the output path. This
   762→   * enables deterministic real Electron E2E testing without native
   763→   * dialog interaction. Production behavior is unchanged when the env
   764→   * var is absent.
   765→   */
   766→  async exportPdf(
   767→    wcId: number,
   768→    callerWindow: BrowserWindow | undefined,
   769→    request: {
   770→      fileName: string
   771→      html: string
   772→      landscape: boolean
   773→      pageSize: SpreadsheetPdfOptions['pageSize']
   774→      margins: SpreadsheetPdfOptions['margins']
   775→      scale: number
   776→    },
   777→  ): Promise<{ canceled: true } | { canceled: false; path: string }> {
   778→    const pdfRenderer = this.deps.pdfRenderer
   779→    if (!pdfRenderer) {
   780→      throw new Error('PDF renderer not available')
   781→    }
   782→
   783→    // 1. Authorize output path (save dialog or test override)
   784→    let outputPath: string
   785→    const testOutPath = process.env['GENOFFICE_PDF_TEST_OUTPATH']
   786→    if (testOutPath !== undefined && testOutPath.length > 0) {
   787→      // Test-only: skip the dialog, use the env var path directly.
   788→      // This is NOT a security bypass — the env var is set only in test
   789→      // environments. Production never sets it.
   790→      outputPath = testOutPath
   791→    } else {
   792→      const dialogOptions = {
   793→        defaultPath: request.fileName,
   794→        filters: [{ name: 'PDF', extensions: ['pdf'] }],
   795→      }
   796→      const selection = callerWindow
   797→        ? await dialog.showSaveDialog(callerWindow, dialogOptions)
   798→        : await dialog.showSaveDialog(dialogOptions)
   799→      if (selection.canceled || !selection.filePath) return { canceled: true }
   800→      outputPath = selection.filePath
   801→    }
   802→
   803→    // 2. Render HTML → PDF bytes (renderer owns hidden BrowserWindow)
   804→    const renderResult = await pdfRenderer.renderToPdf(request.html, {
   805→      landscape: request.landscape,
   806→      pageSize: request.pageSize,
   807→      margins: request.margins,
   808→      scale: request.scale,
   809→    })
   810→
   811→    // 3. If render failed → throw (NO file written)
   812→    if (!renderResult.ok) {
   813→      throw new Error(renderResult.message)
   814→    }
   815→
   816→    // 4. Write PDF bytes to the authorized path
   817→    await writeFile(outputPath, renderResult.data)
   818→    return { canceled: false, path: outputPath }
   819→  }

   821→  async teardown(wcId: number): Promise<void> {
   822→    const state = this.tabs.get(wcId)
   823→    if (!state) return
   824→    state.epoch++
   825→
   826→    const sessionIds = [...state.sessions.keys()]
   827→    await Promise.all(sessionIds.map(async (sid) => {
   828→      await this.withSessionLock(wcId, sid, async () => {
   829→        // If a commit is in progress, wait for it to complete.
   830→        // The mutation lock ensures we don't close the handle while
   831→        // the commit is using it. The commit will check for TEARING_DOWN
   832→        // before entering COMMITTING — but if it's already COMMITTING,
   833→        // we must let it finish (it holds the lock).
   834→        //
   835→        // Since we're inside the lock, the commit has already completed
   836→        // (either successfully or with failure). We can safely close.
   837→        this.setCommitState(wcId, sid, SessionCommitState.TEARING_DOWN)
   838→        await this.closeSession(wcId, sid)
   839→        this.setCommitState(wcId, sid, SessionCommitState.CLOSED)
   840→      })
   841→    }))
   842→    this.tabs.delete(wcId)
   843→  }

   869→  ): Promise<{ openPath: string; suggestSaveAs?: string; csvImport?: boolean; restoreTarget?: string; conversionDir?: string }> {
   870→    const extension = path.slice(path.lastIndexOf('.') + 1).toLowerCase()

   928→/**
   929→ * Reconcile leftover save-commit markers from a previous crash.
   930→ *
   931→ * This must be called once during Sheets main-process startup, BEFORE
   932→ * any migrated workbook operations can begin. It uses app.getPath('userData')
   933→ * to locate the commit marker directory.
   934→ *
   935→ * The function is safe and idempotent — calling it when no markers exist
   936→ * is a no-op.
   937→ *
   938→ * Usage (in sheets-main.ts startup):
   939→ *   import { reconcileSheetsSaveCommits } from './sheets-shell-coordinator'
   940→ *   // ... after app.whenReady():
   941→ *   await reconcileSheetsSaveCommits()
   942→ */
   943→/** Sanitize an AI-provided sheet name into a safe filename base: strip illegal path chars, collapse whitespace, cap length; null if invalid. (Mirrors legacy sanitizeAutoRenameBase.) */
   944→function sanitizeAutoRenameBase(raw: string): string | null {
   945→  const cleaned = raw
   946→    .replace(/[\\/:*?"<>|\u0000-\u001f]/g, ' ')
   947→    .replace(/\s+/g, ' ')
   948→    .trim()
   949→    .replace(/^\.+|\.+$/g, '')
   950→    .trim()
   951→  if (!cleaned) return null
   952→  return cleaned.length > 40 ? cleaned.slice(0, 40).trim() : cleaned
   953→}
